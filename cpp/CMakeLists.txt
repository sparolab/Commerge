cmake_minimum_required(VERSION 3.16)
project(commerge VERSION 2.1.0 LANGUAGES CXX)

# Options
option(COMMERGE_WITH_GUROBI "Use Gurobi when found (GUROBI_HOME or /opt/gurobi*)" ON)
option(COMMERGE_WITH_REGISTRATION "KISS-Matcher + small_gicp registration (thirdparty/KISS-Matcher, small_gicp)" ON)
option(COMMERGE_BUILD_CLI "Build the commerge_cpp command" ON)
option(COMMERGE_BUILD_PYTHON "Build the Python module commerge_cpp (pybind11)" OFF)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

include(FetchContent)
# CMake 4 refuses old dependencies (ROBIN's pmc, cmake_minimum_required < 3.5); children inherit this
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(ENV{CMAKE_POLICY_VERSION_MINIMUM} 3.5)
# never pick up build trees registered by other projects (~/.cmake/packages)
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
set(CMAKE_EXPORT_NO_PACKAGE_REGISTRY ON)
if(CMAKE_VERSION VERSION_GREATER_EQUAL 3.24)
  cmake_policy(SET CMP0135 NEW)
endif()

# Eigen
find_package(Eigen3 3.3 QUIET NO_MODULE)
if(NOT TARGET Eigen3::Eigen)
  FetchContent_Declare(eigen URL https://gitlab.com/libeigen/eigen/-/archive/3.4.0/eigen-3.4.0.tar.gz)
  FetchContent_GetProperties(eigen)
  if(NOT eigen_POPULATED)
    FetchContent_Populate(eigen)
  endif()
  add_library(Eigen3::Eigen INTERFACE IMPORTED)
  set_target_properties(Eigen3::Eigen PROPERTIES INTERFACE_INCLUDE_DIRECTORIES ${eigen_SOURCE_DIR})
endif()

# HiGHS (MIT): system package, else built from source (static)
find_package(highs QUIET CONFIG)
if(NOT TARGET highs::highs)
  message(STATUS "commerge: building HiGHS from source")
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
  set(FAST_BUILD ON CACHE BOOL "" FORCE)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(PYTHON_BUILD_SETUP OFF CACHE BOOL "" FORCE)
  set(ZLIB OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(highs GIT_REPOSITORY https://github.com/ERGO-Code/HiGHS.git GIT_TAG v1.11.0 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(highs)
endif()

add_library(commerge
  src/selection.cpp
  src/solvers.cpp
  src/session.cpp
  src/merge.cpp)
add_library(commerge::commerge ALIAS commerge)
target_include_directories(commerge PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_link_libraries(commerge PUBLIC Eigen3::Eigen PRIVATE highs::highs)

# Gurobi (C API; optional)
if(COMMERGE_WITH_GUROBI)
  file(GLOB _grb_dirs /opt/gurobi*/linux64 /Library/gurobi*/macos_universal2)
  find_path(GUROBI_INCLUDE_DIR gurobi_c.h HINTS $ENV{GUROBI_HOME}/include PATHS ${_grb_dirs} PATH_SUFFIXES include)
  find_library(GUROBI_LIBRARY NAMES gurobi130 gurobi120 gurobi110 gurobi100
               HINTS $ENV{GUROBI_HOME}/lib PATHS ${_grb_dirs} PATH_SUFFIXES lib)
  if(GUROBI_INCLUDE_DIR AND GUROBI_LIBRARY)
    message(STATUS "commerge: Gurobi ${GUROBI_LIBRARY}")
    target_compile_definitions(commerge PRIVATE COMMERGE_WITH_GUROBI)
    target_include_directories(commerge PRIVATE ${GUROBI_INCLUDE_DIR})
    target_link_libraries(commerge PRIVATE ${GUROBI_LIBRARY})
    get_filename_component(GUROBI_LIBDIR ${GUROBI_LIBRARY} DIRECTORY)
    list(APPEND CMAKE_INSTALL_RPATH ${GUROBI_LIBDIR})
  else()
    message(STATUS "commerge: Gurobi not found, HiGHS only")
  endif()
endif()

# Registration: KISS-Matcher (thirdparty submodule) + small_gicp
if(COMMERGE_WITH_REGISTRATION)
  set(_km ${CMAKE_CURRENT_SOURCE_DIR}/../thirdparty/KISS-Matcher/cpp/kiss_matcher)
  find_package(small_gicp QUIET)
  if(NOT TARGET small_gicp::small_gicp)
    message(STATUS "commerge: building small_gicp from source")
    set(BUILD_HELPER ON CACHE BOOL "" FORCE)
    FetchContent_Declare(small_gicp GIT_REPOSITORY https://github.com/koide3/small_gicp.git GIT_TAG v1.0.0 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(small_gicp)
    if(NOT TARGET small_gicp::small_gicp)
      add_library(small_gicp::small_gicp ALIAS small_gicp)
    endif()
  endif()
  # Apple Clang + Homebrew libomp: KISS-Matcher adds -fopenmp but not libomp's headers / library
  find_package(OpenMP QUIET)
  if(APPLE AND TARGET OpenMP::OpenMP_CXX)
    include_directories(${OpenMP_CXX_INCLUDE_DIRS})
    link_libraries(OpenMP::OpenMP_CXX)
  endif()
  if(EXISTS ${_km}/CMakeLists.txt)
    set(USE_SYSTEM_ROBIN OFF CACHE BOOL "" FORCE)
    set(USE_CCACHE OFF CACHE BOOL "" FORCE)
    add_subdirectory(${_km} ${CMAKE_CURRENT_BINARY_DIR}/kiss_matcher EXCLUDE_FROM_ALL)
    target_compile_definitions(commerge PRIVATE COMMERGE_WITH_REGISTRATION)
    target_link_libraries(commerge PRIVATE kiss_matcher::kiss_matcher_core small_gicp::small_gicp)
  else()
    message(WARNING "commerge: thirdparty/KISS-Matcher missing (git submodule update --init); no registration")
  endif()
endif()

if(COMMERGE_BUILD_CLI)
  add_executable(commerge_cpp src/main.cpp)
  target_link_libraries(commerge_cpp PRIVATE commerge)
  install(TARGETS commerge_cpp RUNTIME DESTINATION bin)
endif()

if(COMMERGE_BUILD_PYTHON)
  find_package(Python 3.9 REQUIRED COMPONENTS Interpreter Development.Module)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_core python/bindings.cpp)
  target_link_libraries(_core PRIVATE commerge)
  install(TARGETS _core LIBRARY DESTINATION commerge_cpp)
endif()

if(NOT SKBUILD)
  install(DIRECTORY include/ DESTINATION include)
  install(TARGETS commerge ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
endif()
